function updateSlopeMap(s){
 const legend=$('colorLegend');if(!map||!legend)return;
 if(mapLayers.color){try{map.removeLayer(mapLayers.color)}catch{ }mapLayers.color=null;}
 const r=rasterFor(s);
 legend.hidden=false;
 if(!r||r.kind!=='grid'){
  legend.innerHTML='<b>Carte des pentes : MNT continu requis.</b> Le fond reste le véritable ombrage LiDAR du SPW. Importer un GeoTIFF ou ASC pour calculer des pentes : les sondages espacés ne conviennent pas.';
  return;
 }
 const verts=studyOf(s).polygons.flat();
 if(verts.length<3){legend.textContent='Dessiner une emprise pour cadrer la carte des pentes.';return;}
 const ys=verts.map(p=>p[0]),xs=verts.map(p=>p[1]);
 const ymin=Math.min(...ys),ymax=Math.max(...ys),xmin=Math.min(...xs),xmax=Math.max(...xs);
 const padY=Math.max(.00004,(ymax-ymin)*.1),padX=Math.max(.00005,(xmax-xmin)*.1);
 const south=ymin-padY,north=ymax+padY,west=xmin-padX,east=xmax+padX,n=132;
 const z=new Float32Array(n*n),slopes=new Float32Array(n*n);
 let valid=0;
 for(let y=0;y<n;y++)for(let x=0;x<n;x++){
  const lat=north-(y+.5)*(north-south)/n,lon=west+(x+.5)*(east-west)/n;
  const value=rasterAltitude(lat,lon);z[y*n+x]=value===null?NaN:value;
  if(Number.isFinite(value))valid++;
 }
 if(valid<n*n*.75){legend.textContent='Couverture MNT insuffisante pour produire une carte des pentes fiable sur cette emprise.';return;}
 const dy=(north-south)*111132/n,dx=(east-west)*111320*Math.cos((north+south)*Math.PI/360)/n;
 const canvas=document.createElement('canvas');canvas.width=n;canvas.height=n;
 const ctx=canvas.getContext('2d'),image=ctx.createImageData(n,n);
 let shown=0,min=90,max=0;
 for(let y=1;y<n-1;y++)for(let x=1;x<n-1;x++){
  const i=y*n+x,aa=z[i-1],bb=z[i+1],cc=z[i-n],dd=z[i+n];
  if(![aa,bb,cc,dd].every(Number.isFinite))continue;
  const grad=Math.hypot((bb-aa)/(2*dx),(dd-cc)/(2*dy));
  const deg=Math.atan(grad)*180/Math.PI;slopes[i]=deg;min=Math.min(min,deg);max=Math.max(max,deg);shown++;
  const rgb=deg<5?[81,161,126]:deg<15?[230,204,83]:deg<30?[235,133,60]:[197,65,62];
  const p=i*4;image.data[p]=rgb[0];image.data[p+1]=rgb[1];image.data[p+2]=rgb[2];image.data[p+3]=190;
 }
 if(shown<n*n*.65){legend.textContent='Pentes indisponibles : trop de données altimétriques manquantes.';return;}
 ctx.putImageData(image,0,0);
 mapLayers.color=L.imageOverlay(canvas.toDataURL('image/png'),[[south,west],[north,east]],{opacity:.82,interactive:false}).addTo(map);
 legend.innerHTML=`<b>Pentes calculées sur le MNT importé : ${fmt(min,0)}° à ${fmt(max,0)}°</b><div style="display:flex;height:8px;border-radius:5px;overflow:hidden;margin:6px 0"><i style="background:#51a17e;flex:1"></i><i style="background:#e6cc53;flex:1"></i><i style="background:#eb853c;flex:1"></i><i style="background:#c5413e;flex:1"></i></div><small>Vert &lt; 5° · Jaune 5–15° · Orange 15–30° · Rouge &gt; 30°. Pentes indicatives selon la maille d'échantillonnage, ombrage du sol en fond.</small>`;
}
